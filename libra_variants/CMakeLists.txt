cmake_minimum_required(VERSION 3.20.3)
project(qudec_libra_variants LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(QUDEC_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
add_subdirectory("${QUDEC_ROOT}/libra_pymatching" baseline EXCLUDE_FROM_ALL)
add_library(libra_variant_decoder STATIC "${QUDEC_ROOT}/src/decoder/libra_variant.cpp")
target_link_libraries(libra_variant_decoder PUBLIC qudeclib)
target_compile_options(libra_variant_decoder PRIVATE -O3 -Wall -Wextra)
add_executable(libra_variant_evaluate "${QUDEC_ROOT}/main/libra_variant_evaluate.cpp")
target_link_libraries(libra_variant_evaluate PRIVATE libra_variant_decoder)
enable_testing()
add_executable(libra_variant_test "${QUDEC_ROOT}/tests/libra_variant_test.cpp")
target_link_libraries(libra_variant_test PRIVATE libra_variant_decoder libra_pymatching_decoder)
add_test(NAME libra_variant_preservation_and_syndrome COMMAND libra_variant_test)
