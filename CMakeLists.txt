# Contributor: OpenAI Codex (Libra integration).
cmake_minimum_required(VERSION 3.20.3)
project(qudec VERSION 1.0.0 LANGUAGES CXX)

# initialize compile options
if (CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(COMPILE_OPTIONS
            -ggdb3
            -Wall
            -Wextra
            -Wpedantic
            -Wunused
            )
else()
    set(COMPILE_OPTIONS
            -Ofast
            -flto=auto
            -fstrict-aliasing
            -march=native
            -Wunused
            )
endif()

# Set C++ standard
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

###################################################
# Dependencies
###################################################

# Add subdirectories for dependencies
add_subdirectory(deps/stim)
add_subdirectory(deps/blossom5)
add_subdirectory(deps/PyMatching)

option(ENABLE_MPI "Enable MPI parallelization" OFF)
if(ENABLE_MPI)
    find_package(MPI REQUIRED)
endif()

###################################################
# Tesseract decoder (vendored, minimal)
###################################################
#
# `deps/tesseract` is Tesseract's full upstream tree (its own CMakeLists.txt,
# bazel files, python bindings, tests, ...). We deliberately do NOT
# `add_subdirectory(deps/tesseract)`: its CMakeLists.txt FetchContent's its
# own copy of stim (plus HiGHS, argparse, googletest, pybind11), which would
# collide with the `libstim` target `deps/stim` already defines here (two
# `libstim` targets, two stim ODRs). Instead we compile only the handful of
# translation units the plain `TesseractDecoder` class needs directly into
# `qudeclib`, against our own `libstim` -- the same trick already used by
# `deps/PyMatching/CMakeLists.txt` ("Stim is provided by the parent project").
# HiGHS, argparse, googletest, and pybind11 are for the simplex solver, CLI
# binary, tests, and python bindings respectively -- none are needed here.
#
# OpenAI GPT-6: Tesseract's packed-word bitset removes its Boost download.

find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)

set(TESSERACT_FILES
    deps/tesseract/src/common.cc
    deps/tesseract/src/utils.cc
    deps/tesseract/src/visualization.cc
    deps/tesseract/src/tesseract.cc
)

###################################################
# Source files
###################################################

set(QUDEC_FILES
    src/component_error_distribution.cpp
    src/circuit_generator.cpp
    src/color_code_circuits.cpp
    src/decoder/pymatching.cpp
    src/decoder/blossom5.cpp
    src/decoder/cluster_gap.cpp # Contributor: OpenAI Codex.
    src/decoder/uf.cpp # Contributor: OpenAI Codex (weighted union-find).
    src/decoder/libra.cpp
    src/decoder/harmony.cpp # Contributor: OpenAI Codex (Harmony ensemble).
    src/decoder/seeded.cpp # Contributor: OpenAI Codex.
    # Generated by OpenAI GPT-6-Luna to build the wrapper in its dedicated translation unit.
    src/decoder/tesseract.cpp
    # Contributor: OpenAI Codex (color-code restriction decoder).
    src/decoder/restriction.cpp
    src/decoder/restriction/lattice.cpp
    src/decoder/restriction/spacetime.cpp
    # Contributor: OpenAI Codex (concatenated color-code MWPM).
    src/decoder/concat_mwpm.cpp
    src/decoder/concat_mwpm/matching.cpp
    src/decoder/concat_mwpm/circuit.cpp
    src/decoder/types.cpp
    src/preprocessor/affinity.cpp
    src/globals.cpp
    src/sampler.cpp
    ${TESSERACT_FILES}
)

###################################################
# Quantum Decoder Library
###################################################

add_library(qudeclib SHARED ${QUDEC_FILES})
target_include_directories(qudeclib PUBLIC "src" "deps" "deps/tesseract/src")
target_compile_options(qudeclib PUBLIC ${COMPILE_OPTIONS})
target_link_libraries(qudeclib PUBLIC libstim libpymatching blossom5 Threads::Threads ZLIB::ZLIB)
# Matching scores and rare-event bounds use IEEE infinities and finite checks.
set_source_files_properties(src/decoder/cluster_gap.cpp src/decoder/pymatching.cpp
    src/decoder/blossom5.cpp src/decoder/libra.cpp src/decoder/harmony.cpp src/decoder/uf.cpp src/decoder/seeded.cpp src/sampler.cpp src/preprocessor/affinity.cpp
    PROPERTIES COMPILE_OPTIONS "-fno-fast-math")
set_source_files_properties(src/decoder/restriction.cpp src/decoder/restriction/lattice.cpp
    src/decoder/restriction/spacetime.cpp PROPERTIES COMPILE_OPTIONS "-fno-fast-math")
set_source_files_properties(src/color_code_circuits.cpp src/component_error_distribution.cpp PROPERTIES COMPILE_OPTIONS "-fno-fast-math")
set_source_files_properties(src/decoder/concat_mwpm.cpp src/decoder/concat_mwpm/matching.cpp
    src/decoder/concat_mwpm/circuit.cpp PROPERTIES COMPILE_OPTIONS "-fno-fast-math")
add_library(libra_decoder ALIAS qudeclib)
if(ENABLE_MPI)
    target_compile_definitions(qudeclib PUBLIC ENABLE_MPI)
    target_link_libraries(qudeclib PUBLIC MPI::MPI_CXX)
endif()

###################################################
# Main executable
###################################################

add_executable(qudec main/qudec.cpp)
target_compile_options(qudec PRIVATE ${COMPILE_OPTIONS} -fno-fast-math)
target_link_libraries(qudec PRIVATE qudeclib)

# OpenAI GPT-6-Sol: Build the shared-shot decoder comparison and syndrome recorder.
add_executable(compare_si1000 main/compare_si1000.cpp)
target_compile_options(compare_si1000 PRIVATE ${COMPILE_OPTIONS})

# Contributor: OpenAI Codex (uncorrelated PyMatching complementary-gap study).
add_executable(qudec_eval_gap main/qudec_eval_gap.cpp)

target_link_libraries(qudec_eval_gap PRIVATE qudeclib)

    if(ENABLE_MPI)
