cmake_minimum_required(VERSION 3.20.3)
project(qudec VERSION 1.0.0 LANGUAGES CXX)

# initialize compile options
if (CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(COMPILE_OPTIONS
            -ggdb3
            -Wall
            -Wextra
            -Wpedantic
            -Wunused
            )
else()
    set(COMPILE_OPTIONS
            -Ofast
            -flto=auto
            -fstrict-aliasing
            -march=native
            -Wunused
            )
endif()

# Set C++ standard
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

###################################################
# Dependencies
###################################################

# Add subdirectories for dependencies
add_subdirectory(deps/stim)
add_subdirectory(deps/blossom5)
add_subdirectory(deps/PyMatching)

option(ENABLE_MPI "Enable MPI parallelization" OFF)
if(ENABLE_MPI)
    find_package(MPI REQUIRED)
endif()

###################################################
# Tesseract decoder (vendored, minimal)
###################################################
#
# `deps/tesseract` is Tesseract's full upstream tree (its own CMakeLists.txt,
# bazel files, python bindings, tests, ...). We deliberately do NOT
# `add_subdirectory(deps/tesseract)`: its CMakeLists.txt FetchContent's its
# own copy of stim (plus HiGHS, argparse, googletest, pybind11), which would
# collide with the `libstim` target `deps/stim` already defines here (two
# `libstim` targets, two stim ODRs). Instead we compile only the handful of
# translation units the plain `TesseractDecoder` class needs directly into
# `qudeclib`, against our own `libstim` -- the same trick already used by
# `deps/PyMatching/CMakeLists.txt` ("Stim is provided by the parent project").
# HiGHS, argparse, googletest, and pybind11 are for the simplex solver, CLI
# binary, tests, and python bindings respectively -- none are needed here.
#
# Tesseract does need one more header-only dependency that is not already
# vendored: Boost (`boost::dynamic_bitset`, `boost::hash_range`). It is not
# installed on this system, so we fetch it. We fetch the classic
# (non-modular) release tarball and use it purely as a header tree -- we
# deliberately avoid
# `FetchContent_MakeAvailable(boost)`, which would `add_subdirectory` into
# Boost's own (large, slow) CMake superbuild just to get header-only usage.
find_package(Boost 1.83 QUIET)
if(Boost_FOUND)
    add_library(boost_headers INTERFACE)
    target_link_libraries(boost_headers INTERFACE Boost::boost)
else()
    FetchContent_Declare(
        boost_headers_src
        URL https://archives.boost.io/release/1.83.0/source/boost_1_83_0.tar.gz
        URL_HASH SHA256=c0685b68dd44cc46574cce86c4e17c0f611b15e195be9848dfd0769a0a207628
    )
    FetchContent_GetProperties(boost_headers_src)
    if(NOT boost_headers_src_POPULATED)
        FetchContent_Populate(boost_headers_src)
    endif()
    add_library(boost_headers INTERFACE)
    target_include_directories(boost_headers INTERFACE ${boost_headers_src_SOURCE_DIR})
endif()

find_package(Threads REQUIRED)

set(TESSERACT_FILES
    deps/tesseract/src/common.cc
    deps/tesseract/src/utils.cc
    deps/tesseract/src/visualization.cc
    deps/tesseract/src/tesseract.cc
)

###################################################
# Source files
###################################################

set(QUDEC_FILES
    src/circuit_generator.cpp
    src/decoder/bunchaluts/hypergraph.cpp
    src/decoder/surface_code.cpp
    # Generated by OpenAI GPT-6-Luna to build the wrapper in its dedicated translation unit.
    src/decoder/tesseract.cpp
    src/decoder/surface_code/matching_data.cpp
    src/globals.cpp
    src/sampler.cpp
    ${TESSERACT_FILES}
)

set(BAL_FILES
    src/decoder/bunchaluts/hypergraph.cpp
)


###################################################
# Quantum Decoder Library
###################################################

add_library(qudeclib SHARED ${QUDEC_FILES})
target_include_directories(qudeclib PUBLIC "src" "deps" "deps/tesseract/src")
target_compile_options(qudeclib PUBLIC ${COMPILE_OPTIONS})
target_link_libraries(qudeclib PUBLIC libstim libpymatching blossom5 boost_headers Threads::Threads)
if(ENABLE_MPI)
    target_compile_definitions(qudeclib PUBLIC ENABLE_MPI)
    target_link_libraries(qudeclib PUBLIC MPI::MPI_CXX)
endif()

###################################################
# Main executable
###################################################

add_executable(qudec main/qudec.cpp)
target_compile_options(qudec PRIVATE ${COMPILE_OPTIONS})
target_link_libraries(qudec PRIVATE qudeclib)


add_executable(bal_dem_analysis main/bal_dem_analysis.cpp ${BAL_FILES})
target_compile_options(bal_dem_analysis PRIVATE ${COMPILE_OPTIONS})
target_link_libraries(bal_dem_analysis PRIVATE qudeclib)
