cmake_minimum_required(VERSION 3.20.3)
project(qudec_libra LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(QUDEC_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
# Separate entry point preserves the existing project's build definition.
add_subdirectory("${QUDEC_ROOT}" qudec EXCLUDE_FROM_ALL)
add_library(libra_decoder STATIC "${QUDEC_ROOT}/src/decoder/libra.cpp")
target_link_libraries(libra_decoder PUBLIC qudeclib)
target_compile_options(libra_decoder PRIVATE -O3 -Wall -Wextra)
add_executable(libra_evaluate "${QUDEC_ROOT}/main/libra_evaluate.cpp")
target_link_libraries(libra_evaluate PRIVATE libra_decoder)
add_executable(libra_splitting_evaluate "${QUDEC_ROOT}/main/libra_splitting_evaluate.cpp")
target_link_libraries(libra_splitting_evaluate PRIVATE libra_decoder)
add_executable(libra_splitting_evaluate_fast "${QUDEC_ROOT}/main/libra_splitting_evaluate.cpp")
target_compile_definitions(libra_splitting_evaluate_fast PRIVATE LIBRA_SPLITTING_FAST=1)
target_link_libraries(libra_splitting_evaluate_fast PRIVATE libra_decoder)
add_executable(libra_splitting_evaluate_local "${QUDEC_ROOT}/main/libra_splitting_evaluate.cpp")
target_compile_definitions(libra_splitting_evaluate_local PRIVATE LIBRA_SPLITTING_FAST=1 LIBRA_SPLITTING_LOCAL=1)
target_link_libraries(libra_splitting_evaluate_local PRIVATE libra_decoder)
add_executable(libra_splitting_evaluate_systematic "${QUDEC_ROOT}/main/libra_splitting_evaluate.cpp")
target_compile_definitions(libra_splitting_evaluate_systematic PRIVATE LIBRA_SPLITTING_FAST=1 LIBRA_SPLITTING_LOCAL=1 LIBRA_SPLITTING_SYSTEMATIC=1)
target_link_libraries(libra_splitting_evaluate_systematic PRIVATE libra_decoder)
enable_testing()
add_executable(libra_test "${QUDEC_ROOT}/tests/libra_test.cpp")
target_link_libraries(libra_test PRIVATE libra_decoder)
add_test(NAME libra_small_hypergraph_oracle COMMAND libra_test)
add_executable(libra_rare_event_test "${QUDEC_ROOT}/tests/libra_rare_event_test.cpp")
target_link_libraries(libra_rare_event_test PRIVATE libra_decoder)
add_test(NAME libra_conditional_sampler_oracle COMMAND libra_rare_event_test)
add_executable(libra_splitting_test "${QUDEC_ROOT}/tests/libra_splitting_test.cpp")
target_link_libraries(libra_splitting_test PRIVATE libra_decoder)
add_test(NAME libra_splitting_sampler_oracle COMMAND libra_splitting_test)
add_executable(libra_local_splitting_test "${QUDEC_ROOT}/tests/libra_local_splitting_test.cpp")
target_link_libraries(libra_local_splitting_test PRIVATE libra_decoder)
add_test(NAME libra_local_splitting_sampler_oracle COMMAND libra_local_splitting_test)
