cmake_minimum_required(VERSION 3.20.3)
project(qudec_libra_pymatching LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(QUDEC_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
add_subdirectory("${QUDEC_ROOT}/libra" original EXCLUDE_FROM_ALL)
add_library(libra_pymatching_decoder STATIC "${QUDEC_ROOT}/src/decoder/libra_pymatching.cpp")
target_link_libraries(libra_pymatching_decoder PUBLIC qudeclib)
target_compile_options(libra_pymatching_decoder PRIVATE -O3 -Wall -Wextra)
add_executable(libra_pymatching_evaluate "${QUDEC_ROOT}/main/libra_evaluate.cpp")
target_link_libraries(libra_pymatching_evaluate PRIVATE libra_pymatching_decoder)
add_executable(libra_pymatching_splitting_evaluate "${QUDEC_ROOT}/main/libra_splitting_evaluate.cpp")
target_compile_definitions(libra_pymatching_splitting_evaluate PRIVATE
    LIBRA_SPLITTING_FAST=1 LIBRA_SPLITTING_LOCAL=1 LIBRA_SPLITTING_SYSTEMATIC=1)
target_link_libraries(libra_pymatching_splitting_evaluate PRIVATE libra_pymatching_decoder)
enable_testing()
add_executable(libra_pymatching_test "${QUDEC_ROOT}/tests/libra_test.cpp")
target_link_libraries(libra_pymatching_test PRIVATE libra_pymatching_decoder)
add_test(NAME libra_pymatching_hypergraph_oracle COMMAND libra_pymatching_test)
add_executable(libra_pymatching_assignment_test "${QUDEC_ROOT}/tests/libra_pymatching_assignment_test.cpp")
target_link_libraries(libra_pymatching_assignment_test PRIVATE libra_pymatching_decoder)
add_test(NAME libra_pymatching_assignment_oracle COMMAND libra_pymatching_assignment_test)
